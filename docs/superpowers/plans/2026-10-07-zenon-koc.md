```python
"""Zenon (stil ve bakım PWA'sı) için bildirim servisi. Kullanıcı isteği 2026-10-07.

Seneca'dan yalıtılmıştır: kendi veritabanı (data/zenon.db), kendi uçları (/zenon/api/...), Seneca'nın VAPID
anahtarını yalnızca okur. Sunucu yalnız hatırlatma metnini ve saatini görür; kişisel bakım verisi telefonda kalır.
"""
import hashlib
import hmac
import json
import logging
import sqlite3
from contextlib import contextmanager
from datetime import datetime, timedelta, timezone

from config import DATA_DIR, PUSH_CONTACT

log = logging.getLogger(__name__)
DB = DATA_DIR / "zenon.db"
KOKEN = "https://ross002200.github.io"
ACILIS_URL = KOKEN + "/Ross/#today"
SURUM = 1
AZAMI_CIHAZ, AZAMI_HATIRLATMA = 5, 100
GEC_KALMA, GERI_TOLERANS, ILERI = timedelta(minutes=30), timedelta(minutes=5), timedelta(days=8)


class Hata(Exception):
    def __init__(self, kod, mesaj):
        super().__init__(mesaj)
        self.kod = kod


def _simdi():
    return datetime.now(timezone.utc)


def _iso(d):
    return d.astimezone(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")


@contextmanager
def _baglan():
    c = sqlite3.connect(DB, timeout=10)
    c.row_factory = sqlite3.Row
    try:
        yield c
        c.commit()
    finally:
        c.close()


def kur():
    DB.parent.mkdir(parents=True, exist_ok=True)
    with _baglan() as c:
        c.executescript("""
            CREATE TABLE IF NOT EXISTS cihazlar (cihaz TEXT PRIMARY KEY, token_ozet TEXT NOT NULL, endpoint TEXT NOT NULL,
                p256dh TEXT NOT NULL, auth TEXT NOT NULL, olusturma TEXT NOT NULL, son_senkron TEXT);
            CREATE TABLE IF NOT EXISTS hatirlatmalar (id INTEGER PRIMARY KEY, cihaz TEXT NOT NULL, zaman_utc TEXT NOT NULL,
                baslik TEXT NOT NULL, govde TEXT NOT NULL, etiket TEXT NOT NULL, gonderildi INTEGER NOT NULL DEFAULT 0);
            CREATE INDEX IF NOT EXISTS ix_zenon_hat ON hatirlatmalar (gonderildi, zaman_utc);
        """)


def _ozet(token):
    return hashlib.sha256(token.encode()).hexdigest()


def _kimlik(cihaz, token):
    if not (isinstance(cihaz, str) and isinstance(token, str) and 16 <= len(cihaz) <= 128 and 32 <= len(token) <= 128):
        raise Hata(400, "Geçersiz cihaz ya da anahtar")


def abone(cihaz, token, abonelik):
    _kimlik(cihaz, token)
    try:
        endpoint, p256dh, auth = abonelik["endpoint"], abonelik["keys"]["p256dh"], abonelik["keys"]["auth"]
    except (KeyError, TypeError):
        raise Hata(400, "Geçersiz abonelik")
    if not str(endpoint).startswith("https://"):
        raise Hata(400, "Geçersiz abonelik")
    with _baglan() as c:
        var = c.execute("SELECT token_ozet FROM cihazlar WHERE cihaz = ?", (cihaz,)).fetchone()
        if var:
            if not hmac.compare_digest(var["token_ozet"], _ozet(token)):
                raise Hata(403, "Anahtar eşleşmiyor")
            c.execute("UPDATE cihazlar SET endpoint = ?, p256dh = ?, auth = ? WHERE cihaz = ?", (endpoint, p256dh, auth, cihaz))
        else:
            if c.execute("SELECT COUNT(*) FROM cihazlar").fetchone()[0] >= AZAMI_CIHAZ:
                raise Hata(409, "Cihaz sınırı dolu")
            c.execute("INSERT INTO cihazlar (cihaz, token_ozet, endpoint, p256dh, auth, olusturma) VALUES (?, ?, ?, ?, ?, ?)",
                      (cihaz, _ozet(token), endpoint, p256dh, auth, _iso(_simdi())))
    return {"ok": True}


def dogrula(cihaz, token):
    _kimlik(cihaz, token)
    with _baglan() as c:
        var = c.execute("SELECT token_ozet FROM cihazlar WHERE cihaz = ?", (cihaz,)).fetchone()
    if not var or not hmac.compare_digest(var["token_ozet"], _ozet(token)):
        raise Hata(403, "Yetkisiz")


def _zaman(s):
    try:
        return datetime.fromisoformat(str(s).replace("Z", "+00:00")).astimezone(timezone.utc)
    except ValueError:
        raise Hata(400, "Geçersiz zaman")


def plan_yaz(cihaz, token, liste):
    dogrula(cihaz, token)
    if not isinstance(liste, list) or len(liste) > AZAMI_HATIRLATMA:
        raise Hata(400, "En fazla 100 hatırlatma")
    simdi, satirlar = _simdi(), []
    for h in liste:
        try:
            zaman, baslik, govde, etiket = _zaman(h["at"]), str(h["title"]), str(h["body"]), str(h.get("tag", "zenon"))
        except (KeyError, TypeError):
            raise Hata(400, "Eksik alan")
        if len(baslik) > 80 or len(govde) > 300 or len(etiket) > 64:
            raise Hata(400, "Metin çok uzun")
        if not (simdi - GERI_TOLERANS <= zaman <= simdi + ILERI):
            raise Hata(400, "Zaman aralık dışında")
        satirlar.append((cihaz, _iso(zaman), baslik, govde, etiket))
    with _baglan() as c:
        c.execute("DELETE FROM hatirlatmalar WHERE cihaz = ? AND gonderildi = 0", (cihaz,))
        c.executemany("INSERT INTO hatirlatmalar (cihaz, zaman_utc, baslik, govde, etiket) VALUES (?, ?, ?, ?, ?)", satirlar)
        c.execute("UPDATE cihazlar SET son_senkron = ? WHERE cihaz = ?", (_iso(simdi), cihaz))
    return len(satirlar)


def _push(satir, veri):
    """'ok' | 'sil' (abonelik bitti) | 'hata'."""
    from pywebpush import WebPushException, webpush

    from push import PRIVATE_PEM

    try:
        webpush(subscription_info={"endpoint": satir["endpoint"], "keys": {"p256dh": satir["p256dh"], "auth": satir["auth"]}},
                data=veri, vapid_private_key=str(PRIVATE_PEM), vapid_claims={"sub": PUSH_CONTACT}, ttl=1800, timeout=10)
        return "ok"
    except WebPushException as ex:
        durum = ex.response.status_code if ex.response is not None else None
        if durum in (404, 410):
            return "sil"
        log.warning("Zenon push gönderilemedi: %s", ex)
        return "hata"


def _veri(baslik, govde, etiket):
    return json.dumps({"title": baslik, "body": govde, "tag": etiket, "url": ACILIS_URL}, ensure_ascii=False)


def test_gonder(cihaz, token):
    dogrula(cihaz, token)
    with _baglan() as c:
        satir = c.execute("SELECT * FROM cihazlar WHERE cihaz = ?", (cihaz,)).fetchone()
    return _push(satir, _veri("Zenon", "Bildirimler çalışıyor. Yarın sabah görüşürüz.", "test")) == "ok"


def gonder():
    simdi = _simdi()
    sonuc = {"gonderildi": 0, "atlandi": 0, "silindi": 0}
    with _baglan() as c:
        vakti = c.execute("""SELECT h.id, h.zaman_utc, h.baslik, h.govde, h.etiket, k.cihaz, k.endpoint, k.p256dh, k.auth
                             FROM hatirlatmalar h JOIN cihazlar k ON k.cihaz = h.cihaz
                             WHERE h.gonderildi = 0 AND h.zaman_utc <= ? ORDER BY h.zaman_utc""", (_iso(simdi),)).fetchall()
        for h in vakti:
            if simdi - _zaman(h["zaman_utc"]) > GEC_KALMA:
                c.execute("UPDATE hatirlatmalar SET gonderildi = 2 WHERE id = ?", (h["id"],))
                sonuc["atlandi"] += 1
                continue
            c.execute("UPDATE hatirlatmalar SET gonderildi = 1 WHERE id = ?", (h["id"],))   # önce işaretle: tekrar yok
            durum = _push(h, _veri(h["baslik"], h["govde"], h["etiket"]))
            if durum == "sil":
                c.execute("DELETE FROM hatirlatmalar WHERE cihaz = ?", (h["cihaz"],))
                c.execute("DELETE FROM cihazlar WHERE cihaz = ?", (h["cihaz"],))
                sonuc["silindi"] += 1
            elif durum == "ok":
                sonuc["gonderildi"] += 1
        c.execute("DELETE FROM hatirlatmalar WHERE gonderildi != 0 AND zaman_utc < ?", (_iso(simdi - timedelta(days=30)),))
    return sonuc


def guvenli_gonder():
    try:
        s = gonder()
        if s["gonderildi"] or s["silindi"]:
            log.info("Zenon bildirim: %s", s)
    except Exception:
        log.exception("Zenon bildirim görevi hatası (Seneca etkilenmez)")


def saglik():
    return {"ok": True, "surum": SURUM}
```

```python
# Zenon bildirim servisi (2026-10-07): yalıtılmış; bozulursa yalnız Zenon kapanır, Seneca açılmaya devam eder.
try:
    import zenon_push
    zenon_push.kur()
except Exception as _zx:
    zenon_push = None
    logging.getLogger("zenon").warning("Zenon servisi devre dışı: %s", _zx)


def _zenon(request: Request, govde, durum=200):
    y = JSONResponse(govde, status_code=durum)
    if zenon_push and request.headers.get("origin") == zenon_push.KOKEN:
        y.headers["Access-Control-Allow-Origin"] = zenon_push.KOKEN
        y.headers["Access-Control-Allow-Headers"] = "Content-Type, X-Zenon-Cihaz, X-Zenon-Token"
        y.headers["Access-Control-Allow-Methods"] = "GET, POST, PUT"
        y.headers["Vary"] = "Origin"
    return y


async def _zenon_cagir(request: Request, islem):
    if zenon_push is None:
        return _zenon(request, {"detail": "Zenon servisi kapalı"}, 503)
    try:
        return _zenon(request, await islem())
    except Exception as ex:
        kod = getattr(ex, "kod", 500)
        if kod == 500:
            logging.getLogger("zenon").exception("Zenon uç hatası")
        return _zenon(request, {"detail": str(ex) if kod != 500 else "Sunucu hatası"}, kod)


async def _json(request: Request):
    try:
        return await request.json()
    except Exception:
        raise zenon_push.Hata(400, "Geçersiz JSON")
```
